## 2. Preliminary Home/Workspace Interface Contract (`Proposed`)

This contract replaces the untyped state in [`initPrototype`](file:///C:/Users/PC/Documents/Codex/2026-11-05/so-x20/outputs/tgm-ux-prototype/prototype.js#L14-L21) (`activeSession`, `focusState`, `focusSession`, `captureCount`, `activeReviewKind`) with a **pure TypeScript boundary** between presentation (`Design track`) and state/integrations (`Logic track`).

### What This Contract Intentionally Does Not Decide
- It **does not** choose the SQLite library, table schema, or Electron IPC channel names.
- It **does not** choose the FlowState local transport (named pipe vs. authenticated loopback) or exact memory budget number/unit.
- It **does not** choose the Obsidian folder layout or link syntax (`[[wikilink]]` vs. `[markdown](link.md)`), treating link format and destination as opaque strings/options supplied by the logic layer.

```mermaid
flowchart LR
    subgraph DesignTrack["Design Track (React UI + Synthetic Fixtures)"]
        HomeView["Home Surface"]
        WorkspaceView["Workspace Surface"]
        ReviewModal["Approval & Conflict Modal"]
    end

    subgraph Contract["Shared TypeScript Contract (Pure Types)"]
        ViewState["DashboardViewState (Read-Only Props)"]
        Actions["DashboardActions (Async Intent Dispatch)"]
    end

    subgraph LogicTrack["Logic Track (State + Host Adapters)"]
        Store["Domain State & SQLite Port"]
        AGYPort["AGY Terminal & MCP Port"]
        FlowPort["FlowState Session Port"]
        VaultPort["Obsidian Reviewer Port"]
    end

    HomeView --> Actions
    WorkspaceView --> Actions
    ReviewModal --> Actions
    ViewState --> HomeView
    ViewState --> WorkspaceView
    ViewState --> ReviewModal
    Actions --> Store
    Store --> ViewState
    Store <--> AGYPort
    Store <--> FlowPort
    Store <--> VaultPort
```

### Proposed TypeScript Contract Skeleton (For Review, Not Yet Written to Disk)

```typescript
// Pure domain identifiers — opaque to the UI track
export type GoalId = string;
export type ProjectId = string;
export type SessionId = string;
export type FocusSessionId = string;
export type CaptureId = string;
export type ProposalId = string;

export type SurfaceRoute =
  | { view: "home" }
  | { view: "workspace"; projectId: ProjectId; sessionId: SessionId }
  | { view: "projects" }
  | { view: "learning" }
  | { view: "reviewers" };

// Reflects first-release-plan.md §5 failure and availability states honestly
export type DependencyHealth =
  | { status: "ready" }
  | { status: "unavailable"; reason: string; recoverable: boolean }
  | { status: "error"; message: string; lastAttemptIso: string };

// Maps prototype.js focusState ("idle" | "running" | "paused" | "stopped")
// while adding explicit task completion and external FlowState sync states
export type FocusStatusView =
  | { state: "idle"; flowStateHealth: DependencyHealth }
  | {
      state: "running" | "paused";
      focusSessionId: FocusSessionId;
      taskTitle: string;
      projectId?: ProjectId;
      goalId?: GoalId;
      elapsedSecondsAuthoritative: number; // Always from FlowState, never AI-estimated
      startedVia: "dashboard" | "agy" | "flowstate_direct";
    }
  | {
      state: "stopped_task_open";
      focusSessionId: FocusSessionId;
      taskTitle: string;
      recordedDurationSeconds: number;
      canCompleteTask: boolean;
      canAddManualNote: boolean;
    }
  | {
      state: "task_completed";
      focusSessionId: FocusSessionId;
      taskTitle: string;
      recordedDurationSeconds: number;
    };

export interface SuggestedStepView {
  id: string;
  title: string;
  reason: string; // Required by first-release-plan.md §3.1
  estimatedMinutes?: number;
  energyLevel?: "low" | "medium" | "high";
  goalId: GoalId;
  goalTitle: string;
  projectId?: ProjectId;
  projectTitle?: string;
  sourceProjectsUsed: ProjectId[]; // Transparency for cross-project active memory
}

export interface QuickCaptureItemView {
  id: CaptureId;
  text: string;
  createdAtIso: string;
  linkedProjectId?: ProjectId;
  linkedGoalId?: GoalId;
}

// Covers progress, memory, reviewer, and practice updates with explicit conflict states
export type ReviewProposalKind = "goal_progress" | "project_memory" | "obsidian_reviewer" | "learning_history";

export interface ReviewProposalView {
  id: ProposalId;
  kind: ReviewProposalKind;
  projectId?: ProjectId;
  sessionId?: SessionId;
  title: string;
  proposedContentMarkdown: string;
  uncertaintyNotes: string[]; // Required by first-release-plan.md §3.4 & §5
  sourceContextSummary: string; // Shows what context was available or compacted
  targetDestination?: string; // e.g., relative vault path for Obsidian reviewer
  conflictState:
    | { status: "clean" }
    | { status: "external_edit_detected"; detail: string }
    | { status: "file_collision"; detail: string }
    | { status: "memory_budget_reached"; detail: string }
    | { status: "partial_write_recovery_needed"; detail: string };
}

export interface DashboardViewState {
  route: SurfaceRoute;
  filter: { maxMinutes?: number; energy?: "low" | "medium" | "high" };
  suggestions: {
    status: "loading" | "ready" | "empty_no_active_goals" | "ai_unavailable";
    items: SuggestedStepView[];
    statusMessage?: string;
  };
  lastSession?: {
    sessionId: SessionId;
    title: string;
    projectId: ProjectId;
    projectTitle: string;
  };
  recentCaptures: QuickCaptureItemView[];
  focus: FocusStatusView;
  activeWorkspace?: {
    projectId: ProjectId;
    projectTitle: string;
    folderPath?: string; // Optional: supports folderless study goals
    sessionId: SessionId;
    sessionTitle: string;
    agyTerminalHealth: DependencyHealth;
    pendingProposals: ReviewProposalView[];
  };
  activeModalProposal?: ReviewProposalView;
}

// Intent actions emitted by the UI track and fulfilled by the Logic track
export interface DashboardActions {
  navigate(route: SurfaceRoute): void;
  setSuggestionFilter(filter: { maxMinutes?: number; energy?: "low" | "medium" | "high" }): void;
  selectSuggestion(stepId: string, mode: "open_agy_workspace" | "focus_independently"): Promise<void>;
  editSuggestion(stepId: string, updatedTitle: string): Promise<void>;
  dismissSuggestion(stepId: string): Promise<void>;
  requestAlternativeSuggestions(): Promise<void>;
```
